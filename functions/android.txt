# Android application and library configuration
#
# Qt 5.15 Android support is implemented by Qt5AndroidSupport.cmake, which is
# included by Qt5CoreConfigExtras.cmake as soon as ANDROID_PLATFORM is set.
# That support:
#   * creates the "apk" and "aab" targets;
#   * expects the application target to be a SHARED library;
#   * expects it to be named lib<application-binary>_<abi>.so inside
#     <build-dir>/android-build/libs/<abi>.
#
# That is why an Android app cannot use COMPONENTS_CCR_CONFIGURE_APP directly:
# a desktop app is an executable, while an Android app is a shared library
# loaded by QtActivity. SUFFIX also must not be changed to
# "<version>.bin" because androiddeployqt looks for the ABI suffix.
#
# Expected Android package sources by default:
# <app-root>/android
#   | --> AndroidManifest.xml
#   | --> build.gradle
#   | --> res
#           | --> values
#                   | --> libs.xml
#                   | --> strings.xml
#
# See docs/Android for a minimal example. If <app-root>/android does not exist,
# androiddeployqt falls back to the default package template shipped with Qt.
#
# ========== EXAMPLE OF USAGE: =========
#
# COMPONENTS_CCR_ADD_QT(5 Core Gui Widgets Concurrent Sql)
#
# COMPONENTS_CCR_CONFIGURE_ANDROID_APP(
#     ${APP_TARGET_NAME}
#     ${CMAKE_CURRENT_LIST_DIR}
#     PACKAGE_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/android"
#     VERSION_NAME "${PROJECT_VERSION}"
#     VERSION_CODE 10001
#     MIN_SDK_VERSION 21
#     TARGET_SDK_VERSION 30
#     LIBS
#         Qt5::Widgets
#         Qt5::Concurrent
#         Qt5::Sql
# )
#
# COMPONENTS_CCR_CONFIGURE_ANDROID_LIB(
#     true
#     ${LIB_TARGET_NAME}
#     ${CMAKE_CURRENT_LIST_DIR}
#     Qt5::Widgets
#     Qt5::Concurrent
# )
# COMPONENTS_CCR_SET_LIBRARY_VERSION(${LIB_TARGET_NAME} "1.0.0" true)
#
# Then build the installable APK with:
#     cmake --build <build-dir> --target apk
#
# Both functions fall back to the desktop implementation when the build is not
# an Android build, so they can be called unconditionally.
# ================================================== #


# JSON string escaping for android_deployment_settings.json
# CCR_OUT   --> Quoted JSON string
# CCR_VALUE --> Raw string value
function(COMPONENTS_CCR_ANDROID_JSON_QUOTE CCR_OUT CCR_VALUE)
    file(TO_CMAKE_PATH "${CCR_VALUE}" _value)
    string(REPLACE "\\" "\\\\" _value "${_value}")
    string(REPLACE "\"" "\\\"" _value "${_value}")
    string(REPLACE "\n" "\\n" _value "${_value}")
    string(REPLACE "\r" "\\r" _value "${_value}")
    string(REPLACE "\t" "\\t" _value "${_value}")
    set(${CCR_OUT} "\"${_value}\"" PARENT_SCOPE)
endfunction()


# JSON member setting. CCR_VALUE must be a valid JSON literal.
# CCR_OUT --> Resulting JSON string
# CCR_JSON --> Source JSON string
function(COMPONENTS_CCR_ANDROID_JSON_SET CCR_OUT CCR_JSON CCR_KEY CCR_VALUE)
    string(JSON _json SET "${CCR_JSON}" "${CCR_KEY}" "${CCR_VALUE}")
    set(${CCR_OUT} "${_json}" PARENT_SCOPE)
endfunction()


# JSON string member setting
# CCR_OUT --> Resulting JSON string
# CCR_JSON --> Source JSON string
function(COMPONENTS_CCR_ANDROID_JSON_STRING_SET CCR_OUT CCR_JSON CCR_KEY CCR_VALUE)
    COMPONENTS_CCR_ANDROID_JSON_QUOTE(_quoted "${CCR_VALUE}")
    COMPONENTS_CCR_ANDROID_JSON_SET(_json "${CCR_JSON}" "${CCR_KEY}" "${_quoted}")
    set(${CCR_OUT} "${_json}" PARENT_SCOPE)
endfunction()


# JSON comma-separated list member setting
# CCR_OUT --> Resulting JSON string
# CCR_JSON --> Source JSON string
function(COMPONENTS_CCR_ANDROID_JSON_LIST_SET CCR_OUT CCR_JSON CCR_KEY)
    if (NOT ARGN)
        set(${CCR_OUT} "${CCR_JSON}" PARENT_SCOPE)
        return()
    endif()

    string(REPLACE ";" "," _joined "${ARGN}")
    COMPONENTS_CCR_ANDROID_JSON_STRING_SET(_json "${CCR_JSON}" "${CCR_KEY}" "${_joined}")
    set(${CCR_OUT} "${_json}" PARENT_SCOPE)
endfunction()


# Patch of android_deployment_settings.json generated by Qt5AndroidSupport.
# Qt 5.15 (especially 5.15.2) does not expose all deployment values as CMake
# variables and does not write min/target SDK values into the JSON at all.
# Patching the generated file keeps the Qt template intact and only overrides
# the values owned by a project.
#
# CCR_SETTINGS_FILE      --> Path to android_deployment_settings.json
# ARGN                   --> Named arguments:
#   APPLICATION_BINARY     Application binary base name (target name)
#   PACKAGE_SOURCE_DIR     Android package source directory
#   VERSION_NAME           android-version-name
#   VERSION_CODE           android-version-code
#   MIN_SDK_VERSION        android-min-sdk-version
#   TARGET_SDK_VERSION     android-target-sdk-version
#   QML_ROOT_PATH          qml-root-path
#   EXTRA_LIBS             android-extra-libs
#   EXTRA_PLUGINS          android-extra-plugins
#   DEPLOYMENT_DEPENDENCIES deployment-dependencies
function(COMPONENTS_CCR_ANDROID_PATCH_DEPLOYMENT_SETTINGS CCR_SETTINGS_FILE)
    cmake_parse_arguments(CCR_PATCH
        ""
        "APPLICATION_BINARY;PACKAGE_SOURCE_DIR;VERSION_NAME;VERSION_CODE;MIN_SDK_VERSION;TARGET_SDK_VERSION;QML_ROOT_PATH"
        "EXTRA_LIBS;EXTRA_PLUGINS;DEPLOYMENT_DEPENDENCIES"
        ${ARGN}
    )

    if (NOT EXISTS "${CCR_SETTINGS_FILE}")
        return()
    endif()

    file(READ "${CCR_SETTINGS_FILE}" _json)

    if (CCR_PATCH_APPLICATION_BINARY)
        COMPONENTS_CCR_ANDROID_JSON_STRING_SET(_json "${_json}" "application-binary" "${CCR_PATCH_APPLICATION_BINARY}")
    endif()
    if (CCR_PATCH_PACKAGE_SOURCE_DIR)
        COMPONENTS_CCR_ANDROID_JSON_STRING_SET(_json "${_json}" "android-package-source-directory" "${CCR_PATCH_PACKAGE_SOURCE_DIR}")
    endif()
    if (CCR_PATCH_VERSION_NAME)
        COMPONENTS_CCR_ANDROID_JSON_STRING_SET(_json "${_json}" "android-version-name" "${CCR_PATCH_VERSION_NAME}")
    endif()
    if (CCR_PATCH_VERSION_CODE)
        COMPONENTS_CCR_ANDROID_JSON_STRING_SET(_json "${_json}" "android-version-code" "${CCR_PATCH_VERSION_CODE}")
    endif()
    if (CCR_PATCH_MIN_SDK_VERSION)
        COMPONENTS_CCR_ANDROID_JSON_STRING_SET(_json "${_json}" "android-min-sdk-version" "${CCR_PATCH_MIN_SDK_VERSION}")
    endif()
    if (CCR_PATCH_TARGET_SDK_VERSION)
        COMPONENTS_CCR_ANDROID_JSON_STRING_SET(_json "${_json}" "android-target-sdk-version" "${CCR_PATCH_TARGET_SDK_VERSION}")
    endif()
    if (CCR_PATCH_QML_ROOT_PATH)
        COMPONENTS_CCR_ANDROID_JSON_STRING_SET(_json "${_json}" "qml-root-path" "${CCR_PATCH_QML_ROOT_PATH}")
    endif()

    COMPONENTS_CCR_ANDROID_JSON_LIST_SET(_json "${_json}" "android-extra-libs" ${CCR_PATCH_EXTRA_LIBS})
    COMPONENTS_CCR_ANDROID_JSON_LIST_SET(_json "${_json}" "android-extra-plugins" ${CCR_PATCH_EXTRA_PLUGINS})
    COMPONENTS_CCR_ANDROID_JSON_LIST_SET(_json "${_json}" "deployment-dependencies" ${CCR_PATCH_DEPLOYMENT_DEPENDENCIES})

    file(WRITE "${CCR_SETTINGS_FILE}" "${_json}")
endfunction()


# Common Android application configuration function
#
# CCR_APP_NAME      --> Shared library target name (and application-binary name)
# CCR_APP_ROOT_DIR  --> Root directory of an application
# ARGN              --> Libraries to link and/or named options:
#   PACKAGE_SOURCE_DIR      Android package source directory
#                           (default: <app-root>/android)
#   VERSION_NAME            Android version name (default: PROJECT_VERSION)
#   VERSION_CODE            Android version code (default: derived from
#                           PROJECT_VERSION_MAJOR/MINOR/PATCH or 1)
#   MIN_SDK_VERSION         Android minimum SDK version
#                           (default: ANDROID_MIN_SDK_VERSION or 21)
#   TARGET_SDK_VERSION      Android target SDK version
#                           (default: ANDROID_TARGET_SDK_VERSION or 30)
#   QML_ROOT_PATH           QML root path (default: <app-root>)
#   LIBS                    Libraries to link
#   EXTRA_LIBS              Paths added to android-extra-libs
#   EXTRA_PLUGINS           Paths added to android-extra-plugins
#   DEPLOYMENT_DEPENDENCIES Paths added to deployment-dependencies
function(COMPONENTS_CCR_CONFIGURE_ANDROID_APP CCR_APP_NAME CCR_APP_ROOT_DIR)
    cmake_parse_arguments(CCR_ANDROID_APP
        ""
        "PACKAGE_SOURCE_DIR;VERSION_NAME;VERSION_CODE;MIN_SDK_VERSION;TARGET_SDK_VERSION;QML_ROOT_PATH"
        "LIBS;EXTRA_LIBS;EXTRA_PLUGINS;DEPLOYMENT_DEPENDENCIES"
        ${ARGN}
    )

    set(CCR_ANDROID_APP_LIBRARIES
        ${CCR_ANDROID_APP_LIBS}
        ${CCR_ANDROID_APP_UNPARSED_ARGUMENTS}
    )

    COMPONENTS_CCR_IS_ANDROID(CCR_IS_ANDROID_BUILD)
    if (NOT CCR_IS_ANDROID_BUILD)
        message(STATUS "[COMPONENTS] ${CCR_APP_NAME}: not an Android build, using COMPONENTS_CCR_CONFIGURE_APP")
        COMPONENTS_CCR_CONFIGURE_APP(
            ${CCR_APP_NAME}
            ${CCR_APP_ROOT_DIR}
            ${CCR_ANDROID_APP_LIBRARIES}
        )
        return()
    endif()

    if (NOT ANDROID_ABI)
        message(FATAL_ERROR "[COMPONENTS] ${CCR_APP_NAME}: ANDROID_ABI is not set. Use the Android NDK toolchain file")
        return()
    endif()

    get_filename_component(_app_root_dir "${CCR_APP_ROOT_DIR}" ABSOLUTE)
    if (NOT EXISTS "${_app_root_dir}")
        message(FATAL_ERROR "[COMPONENTS] Invalid ${CCR_APP_NAME} root directory: ${CCR_APP_ROOT_DIR}")
        return()
    endif()

    # Qt 5.15 Android support allows only one application binary per package.
    get_property(_configured_app GLOBAL PROPERTY COMPONENTS_CCR_ANDROID_APP_NAME)
    if (_configured_app AND NOT _configured_app STREQUAL ${CCR_APP_NAME})
        message(FATAL_ERROR "[COMPONENTS] Android package already configured for ${_configured_app}, cannot add ${CCR_APP_NAME}")
        return()
    endif()
    set_property(GLOBAL PROPERTY COMPONENTS_CCR_ANDROID_APP_NAME ${CCR_APP_NAME})

    # Android package sources
    set(_package_source_dir "")
    if (CCR_ANDROID_APP_PACKAGE_SOURCE_DIR)
        get_filename_component(_package_source_dir "${CCR_ANDROID_APP_PACKAGE_SOURCE_DIR}" ABSOLUTE)
        if (NOT EXISTS "${_package_source_dir}/AndroidManifest.xml")
            message(FATAL_ERROR "[COMPONENTS] ${CCR_APP_NAME}: AndroidManifest.xml not found in ${_package_source_dir}")
            return()
        endif()
    elseif (ANDROID_PACKAGE_SOURCE_DIR)
        get_filename_component(_package_source_dir "${ANDROID_PACKAGE_SOURCE_DIR}" ABSOLUTE)
        if (NOT EXISTS "${_package_source_dir}/AndroidManifest.xml")
            message(FATAL_ERROR "[COMPONENTS] ${CCR_APP_NAME}: AndroidManifest.xml not found in ${_package_source_dir}")
            return()
        endif()
    elseif (EXISTS "${_app_root_dir}/android/AndroidManifest.xml")
        set(_package_source_dir "${_app_root_dir}/android")
    else()
        message(STATUS "[COMPONENTS] ${CCR_APP_NAME}: Android package sources not found in ${_app_root_dir}/android, using the Qt default package template")
    endif()

    # Version defaults
    set(_version_name "${CCR_ANDROID_APP_VERSION_NAME}")
    if (NOT _version_name)
        set(_version_name "${ANDROID_VERSION_NAME}")
    endif()
    if (NOT _version_name)
        set(_version_name "${PROJECT_VERSION}")
    endif()

    set(_version_code "${CCR_ANDROID_APP_VERSION_CODE}")
    if (NOT _version_code)
        set(_version_code "${ANDROID_VERSION_CODE}")
    endif()
    if (NOT _version_code)
        if (PROJECT_VERSION_MAJOR MATCHES "^[0-9]+$"
                AND PROJECT_VERSION_MINOR MATCHES "^[0-9]+$"
                AND PROJECT_VERSION_PATCH MATCHES "^[0-9]+$")
            math(EXPR _version_code "${PROJECT_VERSION_MAJOR} * 10000 + ${PROJECT_VERSION_MINOR} * 100 + ${PROJECT_VERSION_PATCH}")
        else()
            set(_version_code "1")
        endif()
    endif()

    set(_min_sdk_version "${CCR_ANDROID_APP_MIN_SDK_VERSION}")
    if (NOT _min_sdk_version)
        if (ANDROID_MIN_SDK_VERSION MATCHES "^[0-9]+$")
            set(_min_sdk_version "${ANDROID_MIN_SDK_VERSION}")
        else()
            set(_min_sdk_version "21")
        endif()
    endif()

    set(_target_sdk_version "${CCR_ANDROID_APP_TARGET_SDK_VERSION}")
    if (NOT _target_sdk_version)
        if (ANDROID_TARGET_SDK_VERSION MATCHES "^[0-9]+$")
            set(_target_sdk_version "${ANDROID_TARGET_SDK_VERSION}")
        else()
            set(_target_sdk_version "30")
        endif()
    endif()

    set(_qml_root_path "${CCR_ANDROID_APP_QML_ROOT_PATH}")
    if (NOT _qml_root_path)
        set(_qml_root_path "${_app_root_dir}")
    endif()

    # Extra deployment values. Keep values already provided by a project.
    set(_extra_libs ${ANDROID_EXTRA_LIBS} ${CCR_ANDROID_APP_EXTRA_LIBS})
    set(_extra_plugins ${ANDROID_EXTRA_PLUGINS} ${CCR_ANDROID_APP_EXTRA_PLUGINS})
    set(_deployment_dependencies
        ${ANDROID_DEPLOYMENT_DEPENDENCIES}
        ${CCR_ANDROID_APP_DEPLOYMENT_DEPENDENCIES}
    )
    if (_extra_libs)
        list(REMOVE_DUPLICATES _extra_libs)
    endif()
    if (_extra_plugins)
        list(REMOVE_DUPLICATES _extra_plugins)
    endif()
    if (_deployment_dependencies)
        list(REMOVE_DUPLICATES _deployment_dependencies)
    endif()

    # Files
    file(GLOB_RECURSE RESOURCES "${_app_root_dir}/resources/*.qrc")
    file(GLOB_RECURSE HEADERS "${_app_root_dir}/src/*.hpp" "${_app_root_dir}/src/*.h")
    file(GLOB_RECURSE SOURCES "${_app_root_dir}/src/*.cpp" "${_app_root_dir}/src/*.c")
    file(GLOB_RECURSE UIFILES "${_app_root_dir}/src/*.ui")
    source_group("Ui Files" FILES ${UIFILES})

    # Qt5AndroidSupport sets CMAKE_LIBRARY_OUTPUT_DIRECTORY for the current
    # build. It is also forwarded to multi-ABI ExternalProject builds, so
    # prefer it over CMAKE_BINARY_DIR (which points to the nested build there).
    set(_app_output_dir "${CMAKE_LIBRARY_OUTPUT_DIRECTORY}")
    if (NOT _app_output_dir)
        set(_app_output_dir "${CMAKE_BINARY_DIR}/android-build/libs/${ANDROID_ABI}")
    endif()
    get_filename_component(_app_output_dir "${_app_output_dir}" ABSOLUTE BASE_DIR "${CMAKE_BINARY_DIR}")

    # Android applications are shared libraries loaded by QtActivity.
    add_library(${CCR_APP_NAME} SHARED
        ${HEADERS} ${SOURCES} ${UIFILES} ${RESOURCES}
    )
    set_target_properties(${CCR_APP_NAME} PROPERTIES
        AUTOMOC ON
        AUTOUIC ON
        AUTORCC ON
        PREFIX "lib"
        SUFFIX "_${ANDROID_ABI}.so"
        LIBRARY_OUTPUT_DIRECTORY "${_app_output_dir}"
    )
    target_link_libraries(${CCR_APP_NAME} PRIVATE ${CCR_ANDROID_APP_LIBRARIES})
    target_compile_definitions(${CCR_APP_NAME} PRIVATE CCR_APP_TARGET_NAME="${CCR_APP_NAME}")

    # For Qt UI forms
    target_include_directories(${CCR_APP_NAME} PRIVATE "${_app_root_dir}/src")

    # Installation
    install(TARGETS ${CCR_APP_NAME}
        RUNTIME     DESTINATION ${INSTALL_DESTINATION_PREFIX}
        LIBRARY     DESTINATION ${INSTALL_DESTINATION_PREFIX}/lib
        ARCHIVE     DESTINATION ${INSTALL_DESTINATION_PREFIX}/lib/static
        INCLUDES    DESTINATION ${INSTALL_DESTINATION_PREFIX}/include
    )

    # Build flag configuring. Android uses the NDK Clang toolchain.
    if(CMAKE_BUILD_TYPE STREQUAL "Debug")
        add_compile_definitions(BUILD_MODE_DEBUG)
        target_compile_options(${CCR_APP_NAME} PRIVATE -g)
    elseif(CMAKE_BUILD_TYPE STREQUAL "Release")
        target_compile_options(${CCR_APP_NAME} PRIVATE -O2 -ffunction-sections -fdata-sections)
        target_link_options(${CCR_APP_NAME} PRIVATE -Wl,--gc-sections)
    endif()

    # Values used by Qt5AndroidSupport if it has not been initialized yet.
    # They are also stored in the parent scope so that a later
    # COMPONENTS_CCR_ADD_QT call generates the JSON with the same values.
    set(ANDROID_PACKAGE_SOURCE_DIR "${_package_source_dir}")
    set(ANDROID_VERSION_NAME "${_version_name}")
    set(ANDROID_VERSION_CODE "${_version_code}")
    set(ANDROID_MIN_SDK_VERSION "${_min_sdk_version}")
    set(ANDROID_TARGET_SDK_VERSION "${_target_sdk_version}")
    set(ANDROID_EXTRA_LIBS ${_extra_libs})
    set(ANDROID_EXTRA_PLUGINS ${_extra_plugins})
    set(ANDROID_DEPLOYMENT_DEPENDENCIES ${_deployment_dependencies})
    set(QT_ANDROID_APPLICATION_BINARY "${CCR_APP_NAME}")

    set(ANDROID_PACKAGE_SOURCE_DIR "${_package_source_dir}" PARENT_SCOPE)
    set(ANDROID_VERSION_NAME "${_version_name}" PARENT_SCOPE)
    set(ANDROID_VERSION_CODE "${_version_code}" PARENT_SCOPE)
    set(ANDROID_MIN_SDK_VERSION "${_min_sdk_version}" PARENT_SCOPE)
    set(ANDROID_TARGET_SDK_VERSION "${_target_sdk_version}" PARENT_SCOPE)
    set(ANDROID_EXTRA_LIBS ${_extra_libs} PARENT_SCOPE)
    set(ANDROID_EXTRA_PLUGINS ${_extra_plugins} PARENT_SCOPE)
    set(ANDROID_DEPLOYMENT_DEPENDENCIES ${_deployment_dependencies} PARENT_SCOPE)
    set(QT_ANDROID_APPLICATION_BINARY "${CCR_APP_NAME}" PARENT_SCOPE)

    # Qt5AndroidSupport writes this file when Qt is found. Patch it because:
    #   * the app target may be different from PROJECT_NAME;
    #   * package/version/SDK values may be set after find_package(Qt5);
    #   * Qt 5.15.2 does not write min/target SDK fields at all.
    set(_settings_file "${CMAKE_BINARY_DIR}/android_deployment_settings.json")
    if (TARGET apk)
        COMPONENTS_CCR_ANDROID_PATCH_DEPLOYMENT_SETTINGS("${_settings_file}"
            APPLICATION_BINARY "${CCR_APP_NAME}"
            PACKAGE_SOURCE_DIR "${_package_source_dir}"
            VERSION_NAME "${_version_name}"
            VERSION_CODE "${_version_code}"
            MIN_SDK_VERSION "${_min_sdk_version}"
            TARGET_SDK_VERSION "${_target_sdk_version}"
            QML_ROOT_PATH "${_qml_root_path}"
            EXTRA_LIBS ${_extra_libs}
            EXTRA_PLUGINS ${_extra_plugins}
            DEPLOYMENT_DEPENDENCIES ${_deployment_dependencies}
        )

        # The Qt "apk" target invokes androiddeployqt but does not depend on the
        # application binary. Build it first.
        add_dependencies(apk ${CCR_APP_NAME})
        if (TARGET aab)
            add_dependencies(aab ${CCR_APP_NAME})
        endif()

        # Qt5AndroidSupport builds non-default ABIs through ExternalProject
        # targets but does not connect them to the package targets.
        foreach(_android_abi armeabi-v7a arm64-v8a x86 x86_64)
            if (TARGET ${PROJECT_NAME}-${_android_abi}-builder)
                add_dependencies(apk ${PROJECT_NAME}-${_android_abi}-builder)
                if (TARGET aab)
                    add_dependencies(aab ${PROJECT_NAME}-${_android_abi}-builder)
                endif()
            endif()
        endforeach()

        message(STATUS "[COMPONENTS] Android APK target 'apk' will build ${CCR_APP_NAME}")
        message(STATUS "[COMPONENTS] APK output: ${CMAKE_BINARY_DIR}/android-build/${PROJECT_NAME}.apk")
    else()
        message(STATUS "[COMPONENTS] ${CCR_APP_NAME}: call COMPONENTS_CCR_ADD_QT to initialize Qt Android APK targets")
    endif()

    message(STATUS "[COMPONENTS] Configured Android application " ${CCR_APP_NAME} " for " ${ANDROID_ABI})
endfunction()


# Common Android library configuration function
#
# Android packages are built from a single application shared library. Static
# libraries are linked into it and need no separate packaging. Shared project
# libraries would have to be added to ANDROID_EXTRA_LIBS for every ABI, so this
# function intentionally builds STATIC libraries on Android.
#
# CCR_IS_LIB_STATIC     --> true / false to use static or shared mode
# CCR_LIBRARY_NAME      --> Library target name to configure
# CCR_LIB_ROOT_DIR      --> Root directory of a library (where sources directory stated)
# ARGN                  --> Libraries that target require
function(COMPONENTS_CCR_CONFIGURE_ANDROID_LIB CCR_IS_LIB_STATIC CCR_LIBRARY_NAME CCR_LIB_ROOT_DIR)
    COMPONENTS_CCR_IS_ANDROID(CCR_IS_ANDROID_BUILD)
    if (NOT CCR_IS_ANDROID_BUILD)
        message(STATUS "[COMPONENTS] ${CCR_LIBRARY_NAME}: not an Android build, using COMPONENTS_CCR_CONFIGURE_LIBRARY")
        COMPONENTS_CCR_CONFIGURE_LIBRARY(
            ${CCR_IS_LIB_STATIC}
            ${CCR_LIBRARY_NAME}
            ${CCR_LIB_ROOT_DIR}
            ${ARGN}
        )
        return()
    endif()

    if (NOT CCR_IS_LIB_STATIC MATCHES "TRUE|FALSE|true|false|ON|on|OFF|off")
        message(FATAL_ERROR "[COMPONENTS] Invalid library type (expected TRUE / FALSE or ON / OFF): " ${CCR_IS_LIB_STATIC})
        return()
    endif()

    if (NOT ${CCR_IS_LIB_STATIC})
        message(WARNING "[COMPONENTS] Android APK packaging: ${CCR_LIBRARY_NAME} will be built as STATIC. Shared libraries must be packaged manually via ANDROID_EXTRA_LIBS")
    endif()

    COMPONENTS_CCR_CONFIGURE_LIBRARY(
        TRUE
        ${CCR_LIBRARY_NAME}
        ${CCR_LIB_ROOT_DIR}
        ${ARGN}
    )
    target_include_directories(${CCR_LIBRARY_NAME} PRIVATE
        "${CCR_LIB_ROOT_DIR}/include"
        "${CCR_LIB_ROOT_DIR}/src"
    )
    set_target_properties(${CCR_LIBRARY_NAME} PROPERTIES POSITION_INDEPENDENT_CODE ON)

    message(STATUS "[COMPONENTS] Configured Android static library " ${CCR_LIBRARY_NAME})
endfunction()
